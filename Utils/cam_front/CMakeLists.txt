cmake_minimum_required(VERSION 3.0.2)
project(cam_front)
add_compile_options(-std=c++14 -Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS cv_bridge image_transport roscpp sensor_msgs)
find_package(OpenCV REQUIRED COMPONENTS core imgproc videoio)
catkin_package(CATKIN_DEPENDS cv_bridge image_transport roscpp sensor_msgs)
include_directories(${catkin_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS})
add_executable(cam_front_node src/cam_front_node.cpp)
target_link_libraries(cam_front_node ${catkin_LIBRARIES} ${OpenCV_LIBRARIES})
install(TARGETS cam_front_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
